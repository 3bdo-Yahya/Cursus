public class ImpactAnalysisService : IImpactAnalysisService {
  public async Task<ImpactResult> GetBlockedCoursesAsync(int studentId, int failedCourseId) {
    var adjList = await BuildAdjacencyAsync();
    var blocked = new List<Course>();
    var queue = new Queue<int>();
    
    queue.Enqueue(failedCourseId);
    while (queue.Count > 0) {
      var current = queue.Dequeue();
      foreach (var neighbor in adjList[current]) {
        if (!blocked.Any(c => c.Id == neighbor)) {
          blocked.Add(await _courseRepo.GetByIdAsync(neighbor));
          queue.Enqueue(neighbor);
        }
      }
    }
    return new ImpactResult { BlockedCourses = blocked };
  }
}
